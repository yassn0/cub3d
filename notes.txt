- voici la fonction qui permet le wall sliding : 

void	move_player(t_data *data, double move_x, double move_y)
{
    double	new_x = data->ray.pos_x + move_x;
    double	new_y = data->ray.pos_y + move_y;
    
    // Vérification SÉPARÉE pour X
    if (data->map[(int)data->ray.pos_y][(int)new_x] != '1')
        data->ray.pos_x = new_x;
        
    // Vérification SÉPARÉE pour Y  
    if (data->map[(int)new_y][(int)data->ray.pos_x] != '1')
        data->ray.pos_y = new_y;
}
